create PROCEDURE [dbo].[USP_CREATEFILE]  
 (  
 @FileName varchar(1000),  
 @FileType varchar(100),  
 @FilePath varchar(1000),  
 @FolderId INT,  
 @UserId INT  
 )  
AS  
BEGIN  
  
  INSERT INTO [dbo].[Files]([FileName],[FileType],[CreatedOn],[FilePath],[FolderId],[UserId])  
  VALUES  (@FileName ,@FileType ,getdate()  ,@FilePath  ,@FolderId  ,@UserId)  

  Select * from Files

  END;


  create PROCEDURE [dbo].[USP_CREATEFOLDER]     
 (    
 @Id Int = 0,  
 @FolderName varchar(500),    
 @UserId INT,    
 @IsParent bit,  
 @ParentId INT,  
 @EntityId INT,
 @mtype varchar(50)
 )    
AS    
BEGIN    
    
  INSERT INTO Folders(FolderName,UserId,IsParent,ParentId,EntityId,module_type,CreatedOn)    
        VALUES(@FolderName,@UserId,@IsParent,@ParentId,@EntityId,@mtype,getDate())    
    
END;
